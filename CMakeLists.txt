cmake_minimum_required(VERSION 3.20)
project(LLECP VERSION 0.2.0 LANGUAGES C CXX)
# Zig's bundled linker resolves explicit target dependencies without GNU rpath-link.
if(CMAKE_CXX_COMPILER MATCHES "zig")
    set(CMAKE_SHARED_LIBRARY_RPATH_LINK_C_FLAG "")
    set(CMAKE_SHARED_LIBRARY_RPATH_LINK_CXX_FLAG "")
    set(CMAKE_EXECUTABLE_RPATH_LINK_C_FLAG "-L")
    set(CMAKE_EXECUTABLE_RPATH_LINK_CXX_FLAG "-L")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
# 程序和库统一放在本地可执行文件系统，避免 /mnt 的挂载执行限制。
if(WIN32)
    set(_ll_output "${CMAKE_BINARY_DIR}/bin")
    set(CMAKE_WINDOWS_EXPORT_ALL_SYMBOLS ON)
else()
    set(_ll_output "/usr/local/bin/LLECT")
endif()
if(MSVC)
    add_compile_options(/utf-8 /bigobj /Zc:__cplusplus)
    add_compile_definitions(NOMINMAX _CRT_SECURE_NO_WARNINGS)
endif()
set(LLECP_OUTPUT_DIRECTORY "${_ll_output}" CACHE PATH "Directory for all compiled programs and libraries")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${LLECP_OUTPUT_DIRECTORY}")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${LLECP_OUTPUT_DIRECTORY}")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${LLECP_OUTPUT_DIRECTORY}")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set(_ll_config_output "${LLECP_OUTPUT_DIRECTORY}")
    if(WIN32 AND NOT config STREQUAL "RELEASE")
        set(_ll_config_output "${LLECP_OUTPUT_DIRECTORY}/${config}")
    endif()
    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${config} "${_ll_config_output}")
    set(CMAKE_LIBRARY_OUTPUT_DIRECTORY_${config} "${_ll_config_output}")
    set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY_${config} "${_ll_config_output}")
endforeach()
set(CMAKE_BUILD_RPATH_USE_ORIGIN ON)
set(CMAKE_BUILD_RPATH "$ORIGIN")
set(CMAKE_INSTALL_RPATH "$ORIGIN")
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE RelWithDebInfo CACHE STRING "Build type" FORCE)
endif()
option(BUILD_TESTING "Build deterministic protocol/controller tests" OFF)
include(CTest)
find_package(Threads REQUIRED)
add_subdirectory(LL_Public)
if(WIN32)
    option(LLECP_BUILD_CONTROLLER "Build Linux controller" OFF)
else()
    option(LLECP_BUILD_CONTROLLER "Build Linux controller" ON)
endif()
option(LLECP_BUILD_CPS "Build graphical engineering station" ${WIN32})
if(LLECP_BUILD_CONTROLLER)
    if(WIN32)
        message(FATAL_ERROR "LL_Controller requires Linux. LL_CPS supports Windows and Linux.")
    endif()
    add_subdirectory(LL_Controller)
endif()
if(LLECP_BUILD_CPS)
    add_subdirectory(LL_CPS)
    if(WIN32)
        set_property(DIRECTORY PROPERTY VS_STARTUP_PROJECT LL_CPS)
    endif()
endif()
